  (defun get-current-org-subheading ()
    (interactive)
    (when (eq major-mode 'org-mode)
      (let ((element (org-element-at-point)))
        (when (eq (car element) 'headline)
          (insert (org-element-property :raw-value element))))))

  (defun get-current-org-subheading ()
    (interactive)
    (when (eq major-mode 'org-mode)
      (let* ((element (org-element-at-point))
             (limit (save-excursion (org-table-end))))
        (while (and (not (eq (car element) 'headline)) (search-forward "|") (< (point) limit))
          (setq element (org-element-at-point)))
        (when (eq (car element) 'headline)
          (insert (org-element-property :raw-value element)))))))

  (defun get-current-org-element ()
    (interactive)
    (org-element-at-point))
