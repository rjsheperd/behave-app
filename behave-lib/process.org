template <typename T, typename = void>
struct is_public_member_function : std::false_type {};

template <typename T>
struct is_public_member_function<T, std::void_t<decltype(&T::operator())>> :
    std::bool_constant<std::is_function_v<decltype(&T::operator())>> {};

// Function template to retrieve the names of public methods for a class
template <typename T>
std::vector<std::string> GetPublicMethods() {
    std::vector<std::string> methods;

    // Loop through all members of the class
    // Check if the member is a public member function
    // Add the name of the member to the list of methods
    // Exclude the special case of the constructor (if applicable)
    for_each_member<T>([&methods](auto member) {
        if constexpr (!std::is_constructor_v<decltype(member)>) {
            if constexpr (is_public_member_function<decltype(member)>::value) {
                methods.push_back(member.name);
            }
        }
    });

    return methods;
}

  template<typename ReturnType, typename... Args>
  class Callable {

    template<typename Func>
    struct FunctionWrapper;

    template<typename R, typename... A>
    struct FunctionWrapper<R(A...)> {
      static ReturnType call(const std::function<R(A...)>& func, Args&&... args) {
        return func(std::forward<Args>(args)...);
      }
    };

    std::unordered_map<std::string, FunctionType> funcs;

  public:
    template<typename Func>
    void addFunction(const std::string& name, Func&& func) {

    } else if (index < args.size()) {
      return std::tuple_cat(std::make_tuple(convert<T>(args.at(index))), toTupleImpl<Args...>(index + 1, args));

  }

  template<std::size_t N, std::size_t remaining, typename T, typename ... Ts>
  struct pack_tuple
  {
    static std::tuple<T, Ts...> execute(std::vector<std::string> v) {

      std::tuple head = { convert<T>(v.at(N - remaining)) };
      return std::tuple_cat(head, pack_tuple<N, remaining - 1, Ts...>::execute(v));

    };
  };

  template<std::size_t N, typename T>
  struct pack_tuple<N, 1, T>
  {
    static std::tuple<T> execute(std::vector<std::string> v) {
      return { convert<T>(v.at(N - 1)) };
    };
  };

  template<typename ... Ts>
  void pack(std::tuple<Ts...> & out, std::vector<std::string> v)
  {
    const size_t N = v.size();
    auto result = pack_tuple<N, Ts...>::execute(v);
    out.swap(result);
  }

    // Discrete Setter
    template<typename ObjType, typename EnumType>
    void wrapperDiscreteSetter(ObjType* obj, void (ObjType::*func)(EnumType), std::string str) {
      // Convert the input string to the desired EnumType
      EnumType enumValue = static_cast<EnumType>(std::stoi(str));

      // Call the member function of the object with the converted EnumType
      (obj->*func)(enumValue);
    }

    template<typename ObjType, typename EnumType>
    std::function<void(void*, std::string)> convertDiscreteSetter(void (ObjType::*func)(EnumType)) {
      return [=](void* obj, std::string str) {
	wrapperDiscreteSetter(static_cast<ObjType*>(obj), func, str);
      };
    }

    // Continous Setter
    template<typename ObjType, typename EnumType>
    void wrapperContinuousSetter(ObjType* obj, void (ObjType::*func)(double, EnumType), double num, std::string str) {
      // Convert the input string to the desired EnumType
      EnumType enumValue = _unitResolver.resolveUnit<EnumType>(str);

      // Call the member function of the object with the converted EnumType
      (obj->*func)(num, enumValue);
    }

      return [=](void* obj, double num, std::string str) {
