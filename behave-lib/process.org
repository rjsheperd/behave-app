  template <typename ReturnType, typename Func, typename... Args>
  ReturnType callFunctionFromString(ReturnType (*func)(Args ...), const std::string& paramStr)
  {
      std::istringstream paramStream(paramStr);
      std::tuple<Args...> tupleParams;

      // Parse parameters from string and store in tuple
      std::tuple<std::decay_t<Args>...> params;
      std::apply([&paramStream](auto&&... args) { ((paramStream >> args), ...); }, params);

      // Call the function with the parsed parameters
      return std::apply(func, params);
  }

// Helper function to check if a member is a public member function pointer
template <typename T, typename = void>
struct is_public_member_function : std::false_type {};

template <typename T>
struct is_public_member_function<T, std::void_t<decltype(&T::operator())>> :
    std::bool_constant<std::is_function_v<decltype(&T::operator())>> {};

// Function template to retrieve the names of public methods for a class
template <typename T>
std::vector<std::string> GetPublicMethods() {
    std::vector<std::string> methods;

    // Loop through all members of the class
    // Check if the member is a public member function
    // Add the name of the member to the list of methods
    // Exclude the special case of the constructor (if applicable)
    for_each_member<T>([&methods](auto member) {
        if constexpr (!std::is_constructor_v<decltype(member)>) {
            if constexpr (is_public_member_function<decltype(member)>::value) {
                methods.push_back(member.name);
            }
        }
    });

    return methods;
}

// Usage example
class MyClass {
public:
    void publicMethod1() {}
    void publicMethod2() {}
private:
    void privateMethod() {}
};

int main() {
    std::vector<std::string> methods = GetPublicMethods<MyClass>();

    for (const auto& method : methods) {
        std::cout << method << std::endl;
    }

    return 0;
}
